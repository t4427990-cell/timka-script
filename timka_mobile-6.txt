    menu.Position = UDim2.fromOffset(
        math.clamp(menu.Position.X.Offset, 0, math.max(0, bounds.X - size.X)),
        math.clamp(menu.Position.Y.Offset, 0, math.max(0, bounds.Y - size.Y))
    )
end

local sightParams = RaycastParams.new()
sightParams.FilterType = Enum.RaycastFilterType.Exclude
sightParams.IgnoreWater = true
local gazeThreshold = math.cos(math.rad(25))

local function clearSight(part, character, origin)
    origin = origin or camera.CFrame.Position
    local direction = part.Position - origin
    if direction.Magnitude < 0.001 then return true end
    local result = Workspace:Raycast(origin, direction, sightParams)
    return result == nil or result.Instance:IsDescendantOf(character)
end

local function looksAtMe(character, myPosition)
    local head = character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart")
    if not head or not head:IsA("BasePart") then return false end
    local direction = myPosition - head.Position
    if direction.Magnitude < 0.001 then return true end
    return head.CFrame.LookVector:Dot(direction.Unit) >= gazeThreshold
end

local function refreshFriendStatus()
    if not friendStatus or not active then return end
    local checked, total, failed = 0, 0, 0
    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= player then
            total = total + 1
            if friendCache[other] ~= nil then checked = checked + 1 end
            if friendFailures[other] then failed = failed + 1 end
        end
    end
    friendStatus.Text = string.format("Друзья: проверено %d / %d", checked, total)
    if failed > 0 then friendStatus.Text = friendStatus.Text .. " · не удалось: " .. failed end
end

local function updateESPColors()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local myHead = character and (character:FindFirstChild("Head") or root)
    for other, highlight in pairs(highlights) do
        if highlight.Parent then
            if gazeESPEnabled and myHead then
                local targetChar = other.Character
                local targetHead = targetChar and (targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("HumanoidRootPart"))
                if targetHead then
                    local dir = myHead.Position - targetHead.Position
                    local dot = dir.Magnitude > 0.001 and targetHead.CFrame.LookVector:Dot(dir.Unit) or 0
                    local canSee = clearSight(myHead, targetChar, targetHead.Position)
                    if dot >= gazeThreshold and canSee then
                        highlight.FillColor = Color3.fromRGB(255, 0, 0)
                        highlight.OutlineColor = Color3.fromRGB(255, 0, 0)
                    else
                        highlight.FillColor = Color3.fromRGB(0, 150, 255)
                        highlight.OutlineColor = Color3.fromRGB(0, 150, 255)
                    end
                end
            else
                local color = espColor
                local friend = greenFriends and friendCache[other] == true
                if friend then color = friendColor
                elseif distanceColors and root then
                    local targetCharacter = other.Character
                    local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
                    if targetRoot and (root.Position - targetRoot.Position).Magnitude > nearDistance then
                        color = farColor
                    end
                end
                highlight.FillColor = color
                highlight.OutlineColor = friend and friendColor or Color3.new(1, 1, 1)
            end
        end
    end
end

local function checkFriend(other)
    if other == player or friendPending[other] then return end
    friendPending[other] = true
    friendFailures[other] = nil
    task.spawn(function()
        for attempt = 1, 3 do
            if not active or other.Parent ~= Players then friendPending[other] = nil; return end
            local ok, result = pcall(function() return player:IsFriendsWithAsync(other.UserId) end)
            if not active or other.Parent ~= Players then friendPending[other] = nil; return end
            if ok and type(result) == "boolean" then
                friendCache[other] = result
                friendPending[other] = nil
                nextTargetScan = 0
                updateESPColors()
                refreshFriendStatus()
                return
            end
            if attempt < 3 then task.wait(2) end
        end
        friendPending[other] = nil
        friendFailures[other] = true
        refreshFriendStatus()
    end)
end

local function removeESP(other)
    if highlights[other] then highlights[other]:Destroy(); highlights[other] = nil end
end
local function addESP(other, character)
    removeESP(other)
    if not active or not espEnabled or other == player or not character then return end
    highlights[other] = new("Highlight", character, {Name = "TimkaESP", Adornee = character, FillColor = espColor, OutlineColor = Color3.new(1, 1, 1), FillTransparency = 0.45, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop})
    updateESPColors()
end
local function watchPlayer(other)
    if other == player or playerConnections[other] then return end
    playerConnections[other] = {
        other.CharacterAdded:Connect(function(character) addESP(other, character) end),
        other.CharacterRemoving:Connect(function() removeESP(other) end),
    }
    checkFriend(other)
    addESP(other, other.Character)
    refreshFriendStatus()
end
connect(Players.PlayerAdded, watchPlayer)
connect(Players.PlayerRemoving, function(other)
    removeESP(other)
    for _, connection in ipairs(playerConnections[other] or {}) do connection:Disconnect() end
    playerConnections[other] = nil
    friendCache[other], friendPending[other], friendFailures[other] = nil, nil, nil
    nextTargetScan = 0
    task.defer(refreshFriendStatus)
end)
for _, other in ipairs(Players:GetPlayers()) do watchPlayer(other) end

local colorTimer = 0
connect(RunService.Heartbeat, function(dt)
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    
    if spinEnabled and root and humanoid and humanoid.Health > 0 then
        humanoid.AutoRotate = false
        root.CFrame = root.CFrame * CFrame.Angles(0, math.rad(spinSpeed * dt), 0)
    elseif not spinEnabled and humanoid then
        humanoid.AutoRotate = true
    end

    colorTimer = colorTimer + dt
    if espEnabled and colorTimer >= 0.15 then
        colorTimer = 0
        updateESPColors()
    end
end)
local function getTarget()
    if not camera then return nil end
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local ownHumanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not root or not ownHumanoid or ownHumanoid.Health <= 0 then return nil end
    local center = camera.ViewportSize * 0.5
    local myHead = character:FindFirstChild("Head") or root
    sightParams.FilterDescendantsInstances = {character}
    local closest, distance = nil, math.huge
    local bestVisible, bestLooking = -1, -1
    for _, other in ipairs(Players:GetPlayers()) do
        local targetCharacter = other ~= player and other.Character
        if targetCharacter and (not ignoreFriends or friendCache[other] == false) then
            local humanoid = targetCharacter:FindFirstChildOfClass("Humanoid")
            local part = targetCharacter:FindFirstChild("Head") or targetCharacter:FindFirstChild("HumanoidRootPart")
            if humanoid and humanoid.Health > 0 and part and part:IsA("BasePart") then
                local point, visible = camera:WorldToViewportPoint(part.Position)
                local inFov = visible and (Vector2.new(point.X, point.Y) - center).Magnitude <= fovRadius
                if aim360 or inFov then
                    local current = (root.Position - part.Position).Magnitude
                    local clear = true
                    if preferVisible or preferLooking then clear = clearSight(part, targetCharacter) end
                    local visibleRank = (preferVisible and clear) and 1 or 0
                    local lookingRank = (preferLooking and clear and looksAtMe(targetCharacter, myHead.Position)) and 1 or 0
                    if visibleRank > bestVisible
                        or (visibleRank == bestVisible and lookingRank > bestLooking)
                        or (visibleRank == bestVisible and lookingRank == bestLooking and current < distance) then
                        closest, distance = part, current
                        bestVisible, bestLooking = visibleRank, lookingRank
                    end
                end
            end
        end
    end
    return closest
end

connect(RunService.RenderStepped, function(dt)
    camera = Workspace.CurrentCamera
    if not camera then return end
    if returning and preAimRotation then
        local desired = CFrame.new(camera.CFrame.Position) * preAimRotation
        camera.CFrame = camera.CFrame:Lerp(desired, 1 - math.exp(-dt / 0.1))
        if (camera.CFrame.LookVector - desired.LookVector).Magnitude < 0.01 then
            camera.CFrame = desired; returning = false; preAimRotation = nil
        end
        return
    end
    if not aimEnabled or os.clock() - lastManualInput < 0.35 then return end
    local now = os.clock()
    if now >= nextTargetScan then
        selectedTarget = getTarget()
        nextTargetScan = now + 0.1
    end
    local target = selectedTarget
    if not target or not target.Parent then return end
    if (target.Position - camera.CFrame.Position).Magnitude < 0.001 then return end
    if not preAimRotation then preAimRotation = camera.CFrame.Rotation end
    local desired = CFrame.lookAt(camera.CFrame.Position, target.Position)
    camera.CFrame = smoothness <= 0 and desired or camera.CFrame:Lerp(desired, math.clamp(dt / smoothness, 0, 1))
end)

local function returnToPoint(character)
    if not savedRoot then return false, "Сначала сохрани точку" end
    if not character or character ~= player.Character or not character.Parent then return false, "Персонаж ещё не готов" end
    local root = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid or humanoid.Health <= 0 then return false, "Дождись возрождения" end
    if humanoid.Sit then return false, "Сначала встань с сиденья" end
    local ok = pcall(function()
        local offset = root.CFrame:ToObjectSpace(character:GetPivot())
        character:PivotTo(savedRoot * offset)
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end)
    return ok, ok and "Возврат к точке выполнен" or "Не получилось переместиться"
end
local function savePoint()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid or humanoid.Health <= 0 then status("Дождись появления персонажа"); return end
    savedRoot = root.CFrame
    if marker then marker:Destroy() end
    marker = new("Part", Workspace, {
        Name = "TimkaSavedPoint", Shape = Enum.PartType.Cylinder,
        Size = Vector3.new(0.08, 3, 3), Anchored = true,
        CanCollide = false, CanTouch = false, CanQuery = false,
        Material = Enum.Material.Neon, Color = cyan, Transparency = 0.35,
        CFrame = CFrame.new(root.Position - Vector3.new(0, humanoid.HipHeight + root.Size.Y * 0.5 - 0.04, 0))
            * CFrame.Angles(0, 0, math.rad(90)),
    })
    local position = savedRoot.Position
    status(string.format("Сохранено: %.1f / %.1f / %.1f", position.X, position.Y, position.Z))
end

end

local isThirdPerson = false
button("Вид: От 1-го лица", function(btn)
    isThirdPerson = not isThirdPerson
    local cam = Workspace.CurrentCamera
    if isThirdPerson then
        cam.MaxZoomDistance = 20
        cam.MinZoomDistance = 10
        btn.Text = "Вид: От 3-го лица"
    else
        cam.MaxZoomDistance = 0.5
        cam.MinZoomDistance = 0.5
        btn.Text = "Вид: От 1-го лица"
    end
end)

do
    local priorityEnabled = true
    local priorityTargets = {}
    local priorityRows = {}
    local listOpen = false
    local normalGetTarget = getTarget
    getTarget = function()
        if not priorityEnabled or not camera then return normalGetTarget() end
        local character = player.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        local ownHumanoid = character and character:FindFirstChildOfClass("Humanoid")
        if not root or not ownHumanoid or ownHumanoid.Health <= 0 then return nil end
        local myHead = character:FindFirstChild("Head") or root
        local center = camera.ViewportSize * 0.5
        sightParams.FilterDescendantsInstances = {character}
        local best, bestLooking, bestDistance = nil, -1, math.huge
        for _, other in ipairs(Players:GetPlayers()) do
            if other ~= player and priorityTargets[other.UserId] and (not ignoreFriends or friendCache[other] == false) then
                local targetCharacter = other.Character
                local humanoid = targetCharacter and targetCharacter:FindFirstChildOfClass("Humanoid")
                local part = targetCharacter and (targetCharacter:FindFirstChild("Head") or targetCharacter:FindFirstChild("HumanoidRootPart"))
                if humanoid and humanoid.Health > 0 and part and part:IsA("BasePart") then
                    local point, onScreen = camera:WorldToViewportPoint(part.Position)
                    local inFov = onScreen and (Vector2.new(point.X, point.Y) - center).Magnitude <= fovRadius
                    if (aim360 or inFov) and clearSight(part, targetCharacter) then
                        local distance = (root.Position - part.Position).Magnitude
                        local looking = (preferLooking and looksAtMe(targetCharacter, myHead.Position)) and 1 or 0
                        if looking > bestLooking or (looking == bestLooking and distance < bestDistance) then
                            best, bestLooking, bestDistance = part, looking, distance
                        end
                    end
                end
            end
        end
        return best or normalGetTarget()
    end

    label("— ВЫБОР ИГРОКОВ ДЛЯ АИМА —", 32)
    toggle("Приоритет выбранных", true, function(value) priorityEnabled = value; selectedTarget, nextTargetScan = nil, 0 end)
    local summary = label("Выбрано: 0", 30)

        local sorted = {}
        for other in pairs(priorityRows) do table.insert(sorted, other) end
        table.sort(sorted, function(a, b) return string.lower(a.Name) < string.lower(b.Name) end)
        local count = 0
        for index, other in ipairs(sorted) do
            local object = priorityRows[other]
            local chosen = priorityTargets[other.UserId] == true
            if chosen then count = count + 1 end
            object.Parent.LayoutOrder = index
            object.Text = (chosen and "[✓] " or "[ ] ") .. other.DisplayName .. "\n@" .. other.Name
            object.BackgroundColor3 = chosen and Color3.fromRGB(62, 66, 135) or Color3.fromRGB(21, 35, 56)
            object.TextColor3 = chosen and cyan or Color3.new(1, 1, 1)
        end
        summary.Text = string.format("Выбрано: %d · игроков: %d", count, #sorted)
        listButton.Text = listOpen and "Скрыть список игроков" or "Показать игроков сервера"
    end

    local function addPriorityPlayer(other)
        if other == player or priorityRows[other] then return end
        local frame = new("Frame", listHolder, {
            Name = "PriorityRow_" .. other.UserId,
            Size = UDim2.new(1, 0, 0, 54),
            BackgroundTransparency = 1, BorderSizePixel = 0,

        local object = new("TextButton", frame, {
            Size = UDim2.new(0.7, -3, 1, 0), Text = "", TextWrapped = true,
        })
        styleButton(object)
        object.TextSize = 12
        local tpButton = new("TextButton", frame, {
            Size = UDim2.new(0.3, -3, 1, 0), Position = UDim2.new(0.7, 3, 0, 0), Text = "ТП",
        })
        styleButton(tpButton)
        priorityRows[other] = object
        
        connect(object.Activated, function()
            priorityTargets[other.UserId] = not priorityTargets[other.UserId] or nil
            selectedTarget, nextTargetScan = nil, 0
            refreshRows()
        end)
        
        connect(tpButton.Activated, function()
            if other.Parent ~= Players then status("ТП: игрок уже вышел") return end
            local myCharacter = player.Character
            local myRoot = myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")
            local myHumanoid = myCharacter and myCharacter:FindFirstChildOfClass("Humanoid")
            if not myCharacter or not myCharacter.Parent or not myRoot or not myRoot:IsA("BasePart") or not myHumanoid or myHumanoid.Health <= 0 then status("ТП: дождись своего возрождения") return end
            if myHumanoid.Sit then status("ТП: сначала встань с сиденья") return end
            local targetCharacter = other.Character
            local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
            if not targetCharacter or not targetCharacter.Parent or not targetRoot or not targetRoot:IsA("BasePart") or not targetRoot:IsDescendantOf(Workspace) then status("ТП: цель не загружена") return end
            local ok, err = pcall(function()
                local destination = targetRoot.CFrame * CFrame.new(4, 0, 0)
                local pivotOffset = myRoot.CFrame:ToObjectSpace(myCharacter:GetPivot())
                myCharacter:PivotTo(destination * pivotOffset)
                myRoot.AssemblyLinearVelocity = Vector3.zero
                myRoot.AssemblyAngularVelocity = Vector3.zero
            end)
            if ok then
                selectedTarget, nextTargetScan = nil, 0
                status("Попытка ТП к @" .. other.Name)
            else
                status("ТП: ошибка")
            end
        end)
        refreshRows()
    end

    connect(Players.PlayerAdded, addPriorityPlayer)
    connect(Players.PlayerRemoving, function(other)
        local object = priorityRows[other]
        if object then
            local frame = object.Parent
            if frame then frame:Destroy() else object:Destroy() end
            priorityRows[other] = nil
        end
        priorityTargets[other.UserId] = nil
        selectedTarget, nextTargetScan = nil, 0
        refreshRows()
    end)
    for _, other in ipairs(Players:GetPlayers()) do addPriorityPlayer(other) end
    refreshRows()
    button("Снять выбор со всех", function() priorityTargets = {}; selectedTarget, nextTargetScan = nil, 0; refreshRows() end)
end

-- ===== ФЛИНГ И БЕЛЫЙ СПИСОК =====
local whiteList = {}
local isFlinging = false

local function doFling(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then return end
    if isFlinging then return end
    isFlinging = true
    local myChar = player.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then isFlinging = false return end
    
    task.spawn(function()
        local targetRoot = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
        if targetRoot then
            myChar:PivotTo(targetRoot.CFrame * CFrame.new(0, 0, 2))
            task.wait(0.1)
        end
        
        local spinConn = RunService.Heartbeat:Connect(function(dt)
            if not isFlinging or not myRoot.Parent then return end
            myRoot.AssemblyAngularVelocity = Vector3.new(0, 9999999, 0)
            myRoot.CFrame = myRoot.CFrame * CFrame.Angles(0, math.rad(9999 * dt), 0)
        end)
        
        task.wait(1.5)
        isFlinging = false
        if spinConn then spinConn:Disconnect() end
        local hum = myChar:FindFirstChildOfClass("Humanoid")
        if hum then hum.Health = 0 end
    end)
end
