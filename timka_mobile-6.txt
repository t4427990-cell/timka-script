-- ЛОГИКА ВЗГЛЯДА И ESP
local sightParams = RaycastParams.new()
sightParams.FilterType = Enum.RaycastFilterType.Exclude
sightParams.IgnoreWater = true
local gazeThreshold = math.cos(math.rad(25))

-- Функция проверки стен (теперь принимает точку старта)
local function clearSight(part, character, origin)
    origin = origin or camera.CFrame.Position
    local direction = part.Position - origin
    if direction.Magnitude < 0.001 then return true end
    local result = Workspace:Raycast(origin, direction, sightParams)
    return result == nil or result.Instance:IsDescendantOf(character)
end

local function looksAtMe(character, myPosition)
    local head = character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart")
    if not head or not head:IsA("BasePart") then return false end
    local direction = myPosition - head.Position
    if direction.Magnitude < 0.001 then return true end
    return head.CFrame.LookVector:Dot(direction.Unit) >= gazeThreshold
end

local function refreshFriendStatus()
    if not friendStatus or not active then return end
    local checked, total, failed = 0, 0, 0
    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= player then
            total = total + 1
            if friendCache[other] ~= nil then checked = checked + 1 end
            if friendFailures[other] then failed = failed + 1 end
        end
    end
    friendStatus.Text = string.format("Друзья: проверено %d / %d", checked, total)
    if failed > 0 then friendStatus.Text = friendStatus.Text .. " · не удалось: " .. failed end
end

local function updateESPColors()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local myHead = character and (character:FindFirstChild("Head") or root)
    
    for other, highlight in pairs(highlights) do
        if highlight.Parent then
            -- РЕЖИМ ВЗГЛЯДА (Красный/Синий)
            if gazeESPEnabled and myHead then
                local targetChar = other.Character
                local targetHead = targetChar and (targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("HumanoidRootPart"))
                
                if targetHead then
                    local dir = myHead.Position - targetHead.Position
                    local dot = dir.Magnitude > 0.001 and targetHead.CFrame.LookVector:Dot(dir.Unit) or 0
                    -- Проверяем стены от головы к голове
                    local canSee = clearSight(myHead, targetChar, targetHead.Position)
                    
                    if dot >= gazeThreshold and canSee then
                        highlight.FillColor = Color3.fromRGB(255, 0, 0) -- Красный (смотрит)
                        highlight.OutlineColor = Color3.fromRGB(255, 0, 0)
                    else
                        highlight.FillColor = Color3.fromRGB(0, 150, 255) -- Синий (не смотрит)
                        highlight.OutlineColor = Color3.fromRGB(0, 150, 255)
                    end
                end
            else
                -- СТАРАЯ ЛОГИКА (Дистанция/Друзья)
                local color = espColor
                local friend = greenFriends and friendCache[other] == true
                if friend then
                    color = friendColor
                elseif distanceColors and root then
                    local targetCharacter = other.Character
                    local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
                    if targetRoot and (root.Position - targetRoot.Position).Magnitude > nearDistance then
                        color = farColor
                    end
                end
                highlight.FillColor = color
                highlight.OutlineColor = friend and friendColor or Color3.new(1, 1, 1)
            end
        end
    end
end

local function checkFriend(other)
    if other == player or friendPending[other] then return end
    friendPending[other] = true
    friendFailures[other] = nil
    task.spawn(function()
        for attempt = 1, 3 do
            if not active or other.Parent ~= Players then friendPending[other] = nil; return end
            local ok, result = pcall(function() return player:IsFriendsWithAsync(other.UserId) end)
            if not active or other.Parent ~= Players then friendPending[other] = nil; return end
            if ok and type(result) == "boolean" then
                friendCache[other] = result
                friendPending[other] = nil
                nextTargetScan = 0
                updateESPColors()
                refreshFriendStatus()
                return
            end
            if attempt < 3 then task.wait(2) end
        end
        friendPending[other] = nil
        friendFailures[other] = true
        refreshFriendStatus()
    end)
end

local function removeESP(other)
    if highlights[other] then highlights[other]:Destroy(); highlights[other] = nil end
end
local function addESP(other, character)
    removeESP(other)
    if not active or not espEnabled or other == player or not character then return end
    highlights[other] = new("Highlight", character, {
        Name = "TimkaESP", Adornee = character, FillColor = espColor,
        OutlineColor = Color3.new(1, 1, 1), FillTransparency = 0.45,
        DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
    })
    updateESPColors()
end
local function watchPlayer(other)
    if other == player or playerConnections[other] then return end
    playerConnections[other] = {
        other.CharacterAdded:Connect(function(character) addESP(other, character) end),
        other.CharacterRemoving:Connect(function() removeESP(other) end),
    }
    checkFriend(other)
    addESP(other, other.Character)
    refreshFriendStatus()
end
