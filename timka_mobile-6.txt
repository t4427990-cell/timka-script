local sightParams = RaycastParams.new()
sightParams.FilterType = Enum.RaycastFilterType.Exclude
sightParams.IgnoreWater = true
local gazeThreshold = math.cos(math.rad(25))

local function clearSight(part, character, origin)
    origin = origin or camera.CFrame.Position
    local direction = part.Position - origin
    if direction.Magnitude < 0.001 then return true end
    local result = Workspace:Raycast(origin, direction, sightParams)
    return result == nil or result.Instance:IsDescendantOf(character)
end

local function looksAtMe(character, myPosition)
    local head = character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart")
    if not head or not head:IsA("BasePart") then return false end
    local direction = myPosition - head.Position
    if direction.Magnitude < 0.001 then return true end
    return head.CFrame.LookVector:Dot(direction.Unit) >= gazeThreshold
end

local function refreshFriendStatus()
    if not friendStatus or not active then return end
    local checked, total, failed = 0, 0, 0
    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= player then
            total = total + 1
            if friendCache[other] ~= nil then checked = checked + 1 end
            if friendFailures[other] then failed = failed + 1 end
        end
    end
    friendStatus.Text = string.format("Друзья: проверено %d / %d", checked, total)
    if failed > 0 then friendStatus.Text = friendStatus.Text .. " · не удалось: " .. failed end
end

local function updateESPColors()
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local myHead = character and (character:FindFirstChild("Head") or root)
    for other, highlight in pairs(highlights) do
        if highlight.Parent then
            if gazeESPEnabled and myHead then
                local targetChar = other.Character
                local targetHead = targetChar and (targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("HumanoidRootPart"))
                if targetHead then
                    local dir = myHead.Position - targetHead.Position
                    local dot = dir.Magnitude > 0.001 and targetHead.CFrame.LookVector:Dot(dir.Unit) or 0
                    local canSee = clearSight(myHead, targetChar, targetHead.Position)
                    if dot >= gazeThreshold and canSee then
                        highlight.FillColor = Color3.fromRGB(255, 0, 0)
                        highlight.OutlineColor = Color3.fromRGB(255, 0, 0)
                    else
                        highlight.FillColor = Color3.fromRGB(0, 150, 255)
                        highlight.OutlineColor = Color3.fromRGB(0, 150, 255)
                    end
                end
            else
                local color = espColor
                local friend = greenFriends and friendCache[other] == true
                if friend then color = friendColor
                elseif distanceColors and root then
                    local targetCharacter = other.Character
                    local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
                    if targetRoot and (root.Position - targetRoot.Position).Magnitude > nearDistance then
                        color = farColor
                    end
                end
                highlight.FillColor = color
                highlight.OutlineColor = friend and friendColor or Color3.new(1, 1, 1)
            end
        end
    end
end

local function checkFriend(other)
    if other == player or friendPending[other] then return end
    friendPending[other] = true
    friendFailures[other] = nil
    task.spawn(function()
        for attempt = 1, 3 do
            if not active or other.Parent ~= Players then friendPending[other] = nil; return end
            local ok, result = pcall(function() return player:IsFriendsWithAsync(other.UserId) end)
            if not active or other.Parent ~= Players then friendPending[other] = nil; return end
            if ok and type(result) == "boolean" then
                friendCache[other] = result
                friendPending[other] = nil
                nextTargetScan = 0
                updateESPColors()
                refreshFriendStatus()
                return
            end
            if attempt < 3 then task.wait(2) end
        end
        friendPending[other] = nil
        friendFailures[other] = true
        refreshFriendStatus()
    end)
end

local function removeESP(other)
    if highlights[other] then highlights[other]:Destroy(); highlights[other] = nil end
end
local function addESP(other, character)
    removeESP(other)
    if not active or not espEnabled or other == player or not character then return end
    highlights[other] = new("Highlight", character, {Name = "TimkaESP", Adornee = character, FillColor = espColor, OutlineColor = Color3.new(1, 1, 1), FillTransparency = 0.45, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop})
    updateESPColors()
end
local function watchPlayer(other)
    if other == player or playerConnections[other] then return end
    playerConnections[other] = {
        other.CharacterAdded:Connect(function(character) addESP(other, character) end),
        other.CharacterRemoving:Connect(function() removeESP(other) end),
    }
    checkFriend(other)
    addESP(other, other.Character)
    refreshFriendStatus()
end
connect(Players.PlayerAdded, watchPlayer)
connect(Players.PlayerRemoving, function(other)
    removeESP(other)
    for _, connection in ipairs(playerConnections[other] or {}) do connection:Disconnect() end
    playerConnections[other] = nil
    friendCache[other], friendPending[other], friendFailures[other] = nil, nil, nil
    nextTargetScan = 0
    task.defer(refreshFriendStatus)
end)
for _, other in ipairs(Players:GetPlayers()) do watchPlayer(other) end

local colorTimer = 0
connect(RunService.Heartbeat, function(dt)
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    
    if spinEnabled and root and humanoid and humanoid.Health > 0 then
        humanoid.AutoRotate = false
        root.CFrame = root.CFrame * CFrame.Angles(0, math.rad(spinSpeed * dt), 0)
    elseif not spinEnabled and humanoid then
        humanoid.AutoRotate = true
    end

    colorTimer = colorTimer + dt
    if espEnabled and colorTimer >= 0.15 then
        colorTimer = 0
        updateESPColors()
    end
end)
local function getTarget()
    if not camera then return nil end
    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local ownHumanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not root or not ownHumanoid or ownHumanoid.Health <= 0 then return nil end
    local center = camera.ViewportSize * 0.5
    local myHead = character:FindFirstChild("Head") or root
    sightParams.FilterDescendantsInstances = {character}
    local closest, distance = nil, math.huge
    local bestVisible, bestLooking = -1, -1
    for _, other in ipairs(Players:GetPlayers()) do
        local targetCharacter = other ~= player and other.Character
        if targetCharacter and (not ignoreFriends or friendCache[other] == false) then
            local humanoid = targetCharacter:FindFirstChildOfClass("Humanoid")
            local part = targetCharacter:FindFirstChild("Head") or targetCharacter:FindFirstChild("HumanoidRootPart")
            if humanoid and humanoid.Health > 0 and part and part:IsA("BasePart") then
                local point, visible = camera:WorldToViewportPoint(part.Position)
                local inFov = visible and (Vector2.new(point.X, point.Y) - center).Magnitude <= fovRadius
                if aim360 or inFov then
                    local current = (root.Position - part.Position).Magnitude
                    local clear = true
                    if preferVisible or preferLooking then clear = clearSight(part, targetCharacter) end
                    local visibleRank = (preferVisible and clear) and 1 or 0
                    local lookingRank = (preferLooking and clear and looksAtMe(targetCharacter, myHead.Position)) and 1 or 0
                    if visibleRank > bestVisible
                        or (visibleRank == bestVisible and lookingRank > bestLooking)
                        or (visibleRank == bestVisible and lookingRank == bestLooking and current < distance) then
                        closest, distance = part, current
                        bestVisible, bestLooking = visibleRank, lookingRank
                    end
                end
            end
        end
    end
    return closest
end

connect(RunService.RenderStepped, function(dt)
    camera = Workspace.CurrentCamera
    if not camera then return end
    if returning and preAimRotation then
        local desired = CFrame.new(camera.CFrame.Position) * preAimRotation
        camera.CFrame = camera.CFrame:Lerp(desired, 1 - math.exp(-dt / 0.1))
        if (camera.CFrame.LookVector - desired.LookVector).Magnitude < 0.01 then
            camera.CFrame = desired; returning = false; preAimRotation = nil
        end
        return
    end
    if not aimEnabled or os.clock() - lastManualInput < 0.35 then return end
    local now = os.clock()
    if now >= nextTargetScan then
        selectedTarget = getTarget()
        nextTargetScan = now + 0.1
    end
    local target = selectedTarget
    if not target or not target.Parent then return end
    if (target.Position - camera.CFrame.Position).Magnitude < 0.001 then return end
    if not preAimRotation then preAimRotation = camera.CFrame.Rotation end
    local desired = CFrame.lookAt(camera.CFrame.Position, target.Position)
    camera.CFrame = smoothness <= 0 and desired or camera.CFrame:Lerp(desired, math.clamp(dt / smoothness, 0, 1))
end)

connect(UIS.JumpRequest, function()
    if not infJumpEnabled or UIS:GetFocusedTextBox() or os.clock() - lastJump < 0.12 then return end
    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if humanoid and humanoid.Health > 0 and not humanoid.Sit then
        lastJump = os.clock()
        humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)

local function applyStats()
    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    if speedChanged then humanoid.WalkSpeed = customSpeed end
    if jumpChanged then humanoid.UseJumpPower = true; humanoid.JumpPower = customJump end
end
local scaleNames = {"BodyHeightScale", "BodyWidthScale", "BodyDepthScale", "HeadScale"}
local function applySize()
    if not sizeChanged then return end
    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    for _, name in ipairs(scaleNames) do
        local value = humanoid:FindFirstChild(name)
        if value and value:IsA("NumberValue") then value.Value = customSize end
    end
end

    if humanoid.Sit then return false, "Сначала встань с сиденья" end
    local ok = pcall(function()
        local offset = root.CFrame:ToObjectSpace(character:GetPivot())
        character:PivotTo(savedRoot * offset)
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end)
    return ok, ok and "Возврат к точке выполнен" or "Не получилось переместиться"
end

    if marker then marker:Destroy() end
    marker = new("Part", Workspace, {

        Size = Vector3.new(0.08, 3, 3), Anchored = true,
        CanCollide = false, CanTouch = false, CanQuery = false,

    local position = savedRoot.Position
    status(string.format("Сохранено: %.1f / %.1f / %.1f", position.X, position.Y, position.Z))
end

local function doFling(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then return end
    if isFlinging then return end
    isFlinging = true
    local myChar = player.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")

    task.spawn(function()
        local targetRoot = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
        if targetRoot then
            myChar:PivotTo(targetRoot.CFrame * CFrame.new(0, 0, 2))
            task.wait(0.1)
        end
        
        local spinConn = RunService.Heartbeat:Connect(function(dt)
            if not isFlinging or not myRoot.Parent then return end
            myRoot.AssemblyAngularVelocity = Vector3.new(0, 9999999, 0)
            myRoot.CFrame = myRoot.CFrame * CFrame.Angles(0, math.rad(9999 * dt), 0)
        end)
        
        task.wait(1.5)
        isFlinging = false
        if spinConn then spinConn:Disconnect() end
        local hum = myChar:FindFirstChildOfClass("Humanoid")
        if hum then hum.Health = 0 end
    end)
end

do
    local flyEnabled, noclipEnabled = false, false
    local flySpeed = 50
    local flyRoot, flyHumanoid, attachment, velocity, orientation
    local originalAutoRotate, failedRoot
    local held, collisions = {}, {}
    local moduleClosed = false
    local flyStatus

    local pad = new("Frame", gui, {Name = "TimkaFlightControls", Size = UDim2.fromOffset(76, 122), Position = UDim2.new(1, -88, 0.5, -61), BackgroundTransparency = 1, Visible = false, ZIndex = 12})
    local function verticalButton(text, y, direction)
        local object = new("TextButton", pad, {Size = UDim2.fromOffset(76, 56), Position = UDim2.fromOffset(0, y), Text = text})
        styleButton(object)
        stroke(object, cyan, 0.2)
        connect(object.InputBegan, function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
                held[input] = direction
            end
        end)
    end
    verticalButton("▲ ВВЕРХ", 0, 1)
    verticalButton("▼ ВНИЗ", 66, -1)
    connect(UIS.InputEnded, function(input) held[input] = nil end)
    connect(UIS.WindowFocusReleased, function() held = {} end)

    local function stopFlight()
        if velocity then velocity:Destroy(); velocity = nil end
        if orientation then orientation:Destroy(); orientation = nil end
        if attachment then attachment:Destroy(); attachment = nil end
        if flyHumanoid and flyHumanoid.Parent and originalAutoRotate ~= nil then flyHumanoid.AutoRotate = originalAutoRotate end
        if flyRoot and flyRoot.Parent then
            flyRoot.AssemblyLinearVelocity = Vector3.zero
            flyRoot.AssemblyAngularVelocity = Vector3.zero
        end
        flyRoot, flyHumanoid, originalAutoRotate = nil, nil, nil
        held, pad.Visible = {}, false
    end

    local function restoreCollisions()
        for part, value in pairs(collisions) do
            if part.Parent then part.CanCollide = value end
        end
        collisions = {}
    end

    local function startFlight(root, humanoid)
        stopFlight()
        flyRoot, flyHumanoid = root, humanoid
        originalAutoRotate = humanoid.AutoRotate
        humanoid.AutoRotate = false
        attachment = new("Attachment", root, {Name = "TimkaFlyAttachment"})
        velocity = new("LinearVelocity", root, {
            Name = "TimkaFlyVelocity", Attachment0 = attachment, RelativeTo = Enum.ActuatorRelativeTo.World,
            VelocityConstraintMode = Enum.VelocityConstraintMode.Vector, ForceLimitsEnabled = false, VectorVelocity = Vector3.zero,
        })
        orientation = new("AlignOrientation", root, {
            Name = "TimkaFlyOrientation", Attachment0 = attachment, Mode = Enum.OrientationAlignmentMode.OneAttachment,
            MaxTorque = 1000000, Responsiveness = 25, CFrame = root.CFrame.Rotation,
        })
        root.AssemblyLinearVelocity = Vector3.zero
        pad.Visible = true
    end

    label("— ПОЛЁТ / NOCLIP —", 30)
    toggle("Полёт", false, function(value)
        flyEnabled = value; failedRoot = nil
        if not value then stopFlight() end
        flyStatus.Text = value and "Полёт включён" or "Полёт выключен"
    end)
    stepper("Скорость полёта", 50, 10, 150, 10, 0, function(value) flySpeed = value end)
    flyStatus = label("Полёт выключен", 30)
    toggle("Noclip / сквозь стены", false, function(value)
        noclipEnabled = value
        if not value then restoreCollisions() end
    end)

    connect(RunService.Stepped, function()
        if not noclipEnabled then return end
        local character = player.Character
        if not character then return end
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                if collisions[part] == nil then collisions[part] = part.CanCollide end
                part.CanCollide = false
            end
        end
    end)
    connect(player.CharacterRemoving, function() stopFlight(); restoreCollisions(); failedRoot = nil end)

    local function key(code) return UIS:IsKeyDown(code) and 1 or 0 end
    connect(RunService.Heartbeat, function()
        if not flyEnabled then return end
        local character = player.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        if not root or not humanoid or humanoid.Health <= 0 or root.Anchored or humanoid.Sit then
            if flyRoot then stopFlight() end
            flyStatus.Text = "Полёт: ожидаю живого персонажа"
            return
        end
        if root == failedRoot then return end
        if root ~= flyRoot or not velocity or not velocity.Parent or not orientation or not orientation.Parent then
            local ok = pcall(startFlight, root, humanoid)
            if not ok then stopFlight(); failedRoot = root; flyStatus.Text = "Полёт недоступен"; return end
        end
        local currentCamera = Workspace.CurrentCamera
        if not currentCamera then velocity.VectorVelocity = Vector3.zero; return end
        local look = currentCamera.CFrame.LookVector
        local forward = Vector3.new(look.X, 0, look.Z)
        if forward.Magnitude < 0.001 then forward = Vector3.new(root.CFrame.LookVector.X, 0, root.CFrame.LookVector.Z) end
        forward = forward.Magnitude > 0.001 and forward.Unit or Vector3.new(0, 0, -1)
        local right = Vector3.new(-forward.Z, 0, forward.X)
        local movement = humanoid.MoveDirection
        movement = Vector3.new(movement.X, 0, movement.Z)
        local vertical = 0
        if not UIS:GetFocusedTextBox() then
            for _, direction in pairs(held) do vertical = vertical + direction end
            if UIS.KeyboardEnabled then
                local x = key(Enum.KeyCode.D) + key(Enum.KeyCode.Right) - key(Enum.KeyCode.A) - key(Enum.KeyCode.Left)
                local z = key(Enum.KeyCode.W) + key(Enum.KeyCode.Up) - key(Enum.KeyCode.S) - key(Enum.KeyCode.Down)
                if x ~= 0 or z ~= 0 then movement = right * x + forward * z end
                vertical = vertical + key(Enum.KeyCode.Space) - math.max(key(Enum.KeyCode.LeftControl), key(Enum.KeyCode.RightControl))
            end
        else movement = Vector3.zero end
        local desired = movement + Vector3.new(0, math.clamp(vertical, -1, 1), 0)
        if desired.Magnitude > 1 then desired = desired.Unit end
        velocity.VectorVelocity = desired * flySpeed
        orientation.CFrame = CFrame.lookAt(Vector3.zero, forward)
        flyStatus.Text = "Полёт включён · " .. flySpeed
    end)

    gui.Destroying:Connect(function()
        if moduleClosed then return end
        moduleClosed = true
        flyEnabled, noclipEnabled = false, false
        stopFlight(); restoreCollisions()
    end)

    label("— ТЕМА МЕНЮ —", 30)
    local themeStatus = label("Тема: голубая", 30)
    local basePrimary, baseSecondary = cyan, violet
    local themeObjects = setmetatable({}, {__mode = "k"})
    local function rememberTheme(object)
        if themeObjects[object] then return end
        local properties = {}
        local function remember(property)
            local value = object[property]
            if value == cyan or value == basePrimary then properties[property] = "primary"
            elseif value == violet or value == baseSecondary then properties[property] = "secondary" end
        end
        if object:IsA("TextLabel") or object:IsA("TextButton") or object:IsA("TextBox") then
            if object.Text ~= "Красный" and object.Text ~= "Зелёный" and object.Text ~= "Синий" then remember("TextColor3") end
        elseif object:IsA("UIStroke") or object:IsA("BasePart") or object:IsA("PointLight") then remember("Color")
        elseif object:IsA("ScrollingFrame") then remember("ScrollBarImageColor3")
        elseif object:IsA("ViewportFrame") then remember("LightColor") end
        if next(properties) then themeObjects[object] = properties end
    end
    local function applyTheme(name, primary, secondary)
        for _, object in ipairs(gui:GetDescendants()) do rememberTheme(object) end
        local oldPrimary, oldSecondary = cyan, violet
        cyan, violet = primary, secondary
        for object, properties in pairs(themeObjects) do
            if object.Parent then
                for property, role in pairs(properties) do
                    local previous = role == "primary" and oldPrimary or oldSecondary
                    local base = role == "primary" and basePrimary or baseSecondary
                    if object[property] == previous or object[property] == base then
                        object[property] = role == "primary" and primary or secondary
                    end
                end
            end
        end
        themeStatus.Text = "Тема: " .. name
    end
    button("Голубая тема", function() applyTheme("голубая", Color3.fromRGB(63, 215, 255), Color3.fromRGB(149, 104, 255)) end)
    button("Неоново-зелёная тема", function() applyTheme("неоново-зелёная", Color3.fromRGB(80, 255, 110), Color3.fromRGB(30, 180, 120)) end)
    button("Фиолетовая тема", function() applyTheme("фиолетовая", Color3.fromRGB(195, 130, 255), Color3.fromRGB(255, 100, 190)) end)
end
